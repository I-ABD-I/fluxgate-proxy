4/// Represents a payload with a prefixed header.
5#[derive(Clone, Debug)]
6pub struct PrefixedPayload(Vec<u8>);

8/// Represents an outbound opaque message in the TLS protocol.
9#[derive(Clone, Debug)]
10pub struct OutboundOpaqueMessage {
11    /// The content type of the message.
12    pub typ: ContentType,
13    /// The protocol version of the message.
14    pub version: ProtocolVersion,
15    /// The payload of the message.
16    pub payload: PrefixedPayload,
17}

19/// Represents an outbound plain message in the TLS protocol.
20#[derive(Clone, Debug, Copy)]
21pub struct OutboundPlainMessage<'a> {
22    /// The content type of the message.
23    pub typ: ContentType,
24    /// The protocol version of the message.
25    pub version: ProtocolVersion,
26    /// The payload of the message.

30impl PrefixedPayload {
31    /// Creates a new `PrefixedPayload` with the specified capacity.
32    ///
33    /// # Arguments
34    /// * `capacity` - The capacity of the payload.
35    ///
36    /// # Returns
37    /// * `Self` - The new `PrefixedPayload` instance.
38    pub fn with_capacity(capacity: usize) -> Self {
39        let mut prefixed = Vec::with_capacity(HEADER_SIZE + capacity);
40        prefixed.resize(HEADER_SIZE, 0);
41        Self(prefixed)
42    }
43
44    /// Extends the payload with the specified slice.
45    ///
46    /// # Arguments
47    /// * `slice` - The slice to extend the payload with.
48    pub fn extend(&mut self, slice: &[u8]) {
49        self.0.extend_from_slice(slice);
50    }
51
52    /// Returns the length of the payload.
53    ///
54    /// # Returns
55    /// * `usize` - The length of the payload.
56    pub fn len(&self) -> usize {
57        self.0.len() - HEADER_SIZE
58    }
59}
60
61impl AsMut<[u8]> for PrefixedPayload {
62    /// Returns a mutable reference to the payload.
63    ///
64    /// # Returns
65    /// * `&mut [u8]` - A mutable reference to the payload.
66    fn as_mut(&mut self) -> &mut [u8] {
67        &mut self.0[HEADER_SIZE..]
68    }
69}
70
71impl OutboundOpaqueMessage {
72    /// Encodes the `OutboundOpaqueMessage` into a byte vector.
73    ///
74    /// # Returns
75    /// * `Vec<u8>` - The encoded byte vector.
76    pub fn encode(self) -> Vec<u8> {
77        let length = self.payload.len() as u16;
78        let mut payload = self.payload.0;
79        payload[0] = self.typ.into();
80        payload[1..3].copy_from_slice(&self.version.to_array());
81        payload[3..5].copy_from_slice(&length.to_be_bytes());
82        payload
83    }
84}

86impl OutboundPlainMessage<'_> {
87    /// Converts the `OutboundPlainMessage` into an `OutboundOpaqueMessage`.
88    ///
89    /// # Returns
90    /// * `OutboundOpaqueMessage` - The converted `OutboundOpaqueMessage`.
91    pub fn to_unencrypted_opaque(self) -> OutboundOpaqueMessage {
92        let mut payload = PrefixedPayload::with_capacity(self.payload.len());
93        payload.extend(self.payload);
94        OutboundOpaqueMessage {
95            typ: self.typ,
96            version: self.version,
97            payload,
98        }
99    }
