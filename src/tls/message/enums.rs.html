168impl NamedCurve {

170    pub fn key_exchange_algorithm(self) -> crypto::kx::KeyExchangeAlgorithm {
171        match u16::from(self) {

217enum_builder! {
218    /// The `SignatureScheme` TLS protocol enum.  Values in this enum are taken
219    /// from the various RFCs covering TLS, and are listed by IANA.
220    /// The `Unknown` item is used when processing unrecognised ordinals.
221    #[repr(u16)]
222    pub enum SignatureScheme {
223        RSA_PKCS1_SHA1 => 0x0201,
224        ECDSA_SHA1_Legacy => 0x0203,
225        RSA_PKCS1_SHA256 => 0x0401,
226        ECDSA_NISTP256_SHA256 => 0x0403,
227        RSA_PKCS1_SHA384 => 0x0501,
228        ECDSA_NISTP384_SHA384 => 0x0503,
229        RSA_PKCS1_SHA512 => 0x0601,
230        ECDSA_NISTP521_SHA512 => 0x0603,
231        RSA_PSS_SHA256 => 0x0804,
232        RSA_PSS_SHA384 => 0x0805,
233        RSA_PSS_SHA512 => 0x0806,
234    }
235}
236
237impl SignatureScheme {
238    /// Returns the signature algorithm associated with this `SignatureScheme`.
239    pub(crate) fn algorithm(&self) -> SignatureAlgorithm {
240        match *self {
241            Self::RSA_PKCS1_SHA1
242            | Self::RSA_PKCS1_SHA256
243            | Self::RSA_PKCS1_SHA384
244            | Self::RSA_PKCS1_SHA512
245            | Self::RSA_PSS_SHA256
246            | Self::RSA_PSS_SHA384
247            | Self::RSA_PSS_SHA512 => SignatureAlgorithm::rsa,
248            Self::ECDSA_SHA1_Legacy
249            | Self::ECDSA_NISTP256_SHA256
250            | Self::ECDSA_NISTP384_SHA384
251            | Self::ECDSA_NISTP521_SHA512 => SignatureAlgorithm::ecdsa,
252            _ => SignatureAlgorithm::Unknown(0),
253        }
254    }
