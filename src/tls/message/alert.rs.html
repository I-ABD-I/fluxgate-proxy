44/// Represents the payload of an alert message in TLS.
45#[derive(Debug, Clone)]
46pub struct AlertPayload {
47    /// The level of the alert.
48    pub(crate) level: AlertLevel,
49    /// The description of the alert.
50    pub(crate) description: AlertDescription,
51}
52
53impl Codec<'_> for AlertPayload {
54    /// Encodes the alert payload into a byte vector.
55    ///
56    /// # Arguments
57    ///
58    /// * `bytes` - A mutable reference to a byte vector where the encoded payload will be stored.
59    fn encode(&self, bytes: &mut Vec<u8>) {
60        self.level.encode(bytes);
61        self.description.encode(bytes);
62    }
63
64    /// Reads an alert payload from a byte reader.
65    ///
66    /// # Arguments
67    ///
68    /// * `r` - A mutable reference to a byte reader.
69    ///
70    /// # Returns
71    ///
72    /// A result containing the alert payload or an invalid message error.
73    fn read(r: &mut Reader<'_>) -> Result<Self, InvalidMessage> {
74        let level = AlertLevel::read(r)?;
75        let description = AlertDescription::read(r)?;
76        r.expect_empty("AlertPayload")
77            .map(|_| Self { level, description })
78    }
