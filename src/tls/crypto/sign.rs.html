55/// Attempts to create a signing key from the provided DER-encoded private key.
56///
57/// # Arguments
58///
59/// * `der` - A reference to a `PrivateKeyDer`.
60///
61/// # Returns
62///
63/// A result containing an `Arc` to a `SigningKey` or an error.
64pub fn any_supported_type(der: &PrivateKeyDer<'_>) -> Result<Arc<dyn SigningKey>, Error> {
65    if let Ok(rsa) = RSASigningKey::new(der) {
66        return Ok(Arc::new(rsa));
67    }
68
69    Err(Error::General("Unsupported key type"))
70}

72/// A structure representing an RSA signing key.
73///
74/// This structure provides methods to create a new RSA signing key and to implement the `SigningKey` trait.
75#[derive(Debug)]
76pub struct RSASigningKey {
77    key: Arc<RsaKeyPair>,
78}
79
80impl RSASigningKey {
81    /// Creates a new `RSASigningKey` from the provided DER-encoded private key.
82    ///
83    /// # Arguments
84    ///
85    /// * `der` - A reference to a `PrivateKeyDer`.
86    ///
87    /// # Returns
88    ///
89    /// A result containing an `RSASigningKey` or an error.
90    pub fn new(der: &PrivateKeyDer<'_>) -> Result<Self, Error> {
91        let pair = match der {
92            PrivateKeyDer::Pkcs1(pkcs1) => RsaKeyPair::from_der(pkcs1.secret_pkcs1_der()),
93            PrivateKeyDer::Pkcs8(pkcs8) => RsaKeyPair::from_pkcs8(pkcs8.secret_pkcs8_der()),
94            _ => return Err(Error::General("Invalid RSA key")),
95        }
96        .map_err(|_| Error::General("Invalid RSA key"))?;
97
98        Ok(Self {
99            key: Arc::new(pair),
100        })
101    }
102}

104impl SigningKey for RSASigningKey {
105    /// Chooses a signature scheme from the offered schemes.
106    ///
107    /// # Arguments
108    ///
109    /// * `offered` - A slice of offered `SignatureScheme`s.
110    ///
111    /// # Returns
112    ///
113    /// An optional boxed `Signer`.
114    fn choose_scheme(&self, offered: &[SignatureScheme]) -> Option<Box<dyn Signer>> {
115        RSA_SCHEMES
116            .iter()
117            .find(|scheme| offered.contains(scheme))
118            .map(|scheme| Box::new(RSASigner::new(self.key.clone(), *scheme)) as Box<dyn Signer>)
119    }
120
121    /// Returns the signature algorithm.
122    ///
123    /// # Returns
124    ///
125    /// The `SignatureAlgorithm` used.
126    fn algorithm(&self) -> SignatureAlgorithm {
127        SignatureAlgorithm::rsa
128    }
129}

131/// A structure representing an RSA signer.
132///
133/// This structure provides methods to create a new RSA signer and to implement the `Signer` trait.
134pub struct RSASigner {
135    key: Arc<RsaKeyPair>,
136    scheme: SignatureScheme,
137    encoding: &'static dyn RsaEncoding,
138}
139
140impl RSASigner {
141    /// Creates a new `RSASigner` with the provided key and scheme.
142    ///
143    /// # Arguments
144    ///
145    /// * `key` - An `Arc` to an `RsaKeyPair`.
146    /// * `scheme` - The `SignatureScheme` to use.
147    ///
148    /// # Returns
149    ///
150    /// A new `RSASigner` instance.
151    fn new(key: Arc<RsaKeyPair>, scheme: SignatureScheme) -> Self {
152        use ring::signature;
153
154        let encoding: &dyn signature::RsaEncoding = match scheme {
155            SignatureScheme::RSA_PKCS1_SHA256 => &signature::RSA_PKCS1_SHA256,
156            SignatureScheme::RSA_PKCS1_SHA384 => &signature::RSA_PKCS1_SHA384,
157            SignatureScheme::RSA_PKCS1_SHA512 => &signature::RSA_PKCS1_SHA512,
158            SignatureScheme::RSA_PSS_SHA256 => &signature::RSA_PSS_SHA256,
159            SignatureScheme::RSA_PSS_SHA384 => &signature::RSA_PSS_SHA384,
160            SignatureScheme::RSA_PSS_SHA512 => &signature::RSA_PSS_SHA512,
161            _ => unreachable!(),
162        };
163
164        Self {
165            key,
166            scheme,
167            encoding,
168        }
169    }
170}
171
172impl Signer for RSASigner {
173    /// Signs the provided message.
174    ///
175    /// # Arguments
176    ///
177    /// * `message` - A slice of bytes representing the message to sign.
178    ///
179    /// # Returns
180    ///
181    /// A result containing a vector of bytes or an error.
182    fn sign(&self, message: &[u8]) -> Result<Vec<u8>, Error> {
183        let mut sig = vec![0; self.key.public().modulus_len()];
184        let rng = ring::rand::SystemRandom::new();
185        self.key
186            .sign(self.encoding, &rng, message, &mut sig)
187            .map(|_| sig)
188            .map_err(|_| Error::General("Signing failed"))
189    }
190
191    /// Returns the signature scheme.
192    ///
193    /// # Returns
194    ///
195    /// The `SignatureScheme` used.
196    fn scheme(&self) -> SignatureScheme {
197        self.scheme
198    }
