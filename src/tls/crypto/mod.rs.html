20/// A trait representing a secure random number generator.
21///
22/// This trait is used to fill a destination buffer with random bytes.
23/// It is required to be thread-safe (`Send` and `Sync`).
24pub trait SecureRandom: Send + Sync {
25    /// Fills the provided destination buffer with random bytes.
26    ///
27    /// # Arguments
28    ///
29    /// * `dest` - A mutable slice where the random bytes will be written.
30    ///
31    /// # Errors
32    ///
33    /// Returns a `GetRandomFailed` error if the random number generation fails.
34    fn fill(&self, dest: &mut [u8]) -> Result<(), GetRandomFailed>;
35}

53/// A structure representing a cipher suite.
54///
55/// This structure contains various cryptographic algorithms and parameters
56/// used in a cipher suite.
57#[derive(Clone, Copy, Debug)]
58pub struct CipherSuite {
59    /// The cipher suite identifier.
60    pub suite: message::enums::CipherSuite,
61    /// The hash provider.
62    pub hash_provider: &'static dyn Hash,
63    /// The key exchange algorithm.
64    pub kx: kx::KeyExchangeAlgorithm,
65    /// The supported signature schemes.
66    pub sign: &'static [SignatureScheme],
67    /// The AEAD algorithm.
68    pub aead_algo: &'static dyn AeadAlgorithm,
69    /// The PRF provider.
70    pub prf_provider: &'static dyn Prf,
71}
72
73impl CipherSuite {
74    /// Checks if the cipher suite is usable for a given signature algorithm.
75    ///
76    /// # Arguments
77    ///
78    /// * `sigalg` - The signature algorithm to check.
79    ///
80    /// # Returns
81    ///
82    /// `true` if the cipher suite is usable for the given signature algorithm, `false` otherwise.
83    pub fn usable_for_signature_algorithm(&self, sigalg: SignatureAlgorithm) -> bool {
84        self.sign.iter().any(|scheme| scheme.algorithm() == sigalg)
85    }
86
87    /// Checks if the cipher suite is usable for a given key exchange algorithm.
88    ///
89    /// # Arguments
90    ///
91    /// * `kxalg` - The key exchange algorithm to check.
92    ///
93    /// # Returns
94    ///
95    /// `true` if the cipher suite is usable for the given key exchange algorithm, `false` otherwise.
96    pub fn usable_for_kx_algorithm(&self, kxalg: &kx::KeyExchangeAlgorithm) -> bool {
97        self.kx == *kxalg
98    }
99
100    /// Resolves the supported signature schemes from the offered ones.
101    ///
102    /// # Arguments
103    ///
104    /// * `offered` - A slice of offered signature schemes.
105    ///
106    /// # Returns
107    ///
108    /// A vector of supported signature schemes.
109    pub fn resolve_sig_schemes(&self, offered: &[SignatureScheme]) -> Vec<SignatureScheme> {
110        self.sign
111            .iter()
112            .filter(|pref| offered.contains(pref))
113            .cloned()
114            .collect()
115    }
116}

118/// Checks if a signature scheme is compatible with the given cipher suites.
119///
120/// # Arguments
121///
122/// * `sigscheme` - The signature scheme to check.
123/// * `common_suites` - A slice of supported cipher suites.
124///
125/// # Returns
126///
127/// `true` if the signature scheme is compatible with any of the given cipher suites, `false` otherwise.
128pub(crate) fn compatible_sigscheme_for_suites(
129    sigscheme: SignatureScheme,
130    common_suites: &[SupportedCipherSuite],
131) -> bool {
132    let sigalg = sigscheme.algorithm();
133    common_suites
134        .iter()
135        .any(|&suite| suite.0.usable_for_signature_algorithm(sigalg))
136}
