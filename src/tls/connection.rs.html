31/// Represents the state of a TLS connection.
32pub struct TlsState {
33    /// Indicates if application data can be sent.
34    may_send_appdata: bool,
35    /// Indicates if application data can be received.
36    may_recv_appdata: bool,
37    /// Buffer for sendable TLS data.
38    pub(crate) sendable_tls: Vec<u8>,
39    /// Buffer for received plaintext data.
40    pub(crate) received_plaintext: Vec<u8>,
41    /// Indicates if a close notification has been received.
42    has_received_close: bool,
43    /// Key exchange state.
44    pub(crate) kx_state: KxState,
45    /// Indicates if EOF has been seen.
46    has_seen_eof: bool,
47    /// Record layer for managing TLS records.
48    pub(crate) record_layer: RecordLayer,
49    /// Fragmenter for handling message fragmentation.
50    fragmenter: MessageFragmenter,
51    /// Server Name Indication (SNI) value.
52    pub(crate) sni: Option<DnsName<'static>>,
53}

55/// Represents the state of key exchange.
56pub(crate) enum KxState {
57    /// No key exchange.
58    None,
59    /// Key exchange started with the specified group.
60    Start(&'static dyn SupportedKxGroup),
61    /// Key exchange completed with the specified group.
62    Done(&'static dyn SupportedKxGroup),
63}
64
65impl KxState {
66    /// Marks the key exchange as done.
67    pub(crate) fn done(&mut self) {
68        if let Self::Start(group) = self {
69            *self = Self::Done(*group);
70        }
71    }
72}

98    /// Checks if the connection is in the handshaking state.
99    ///
100    /// # Returns
101    /// `true` if handshaking, `false` otherwise.
102    pub(crate) fn is_handshaking(&self) -> bool {
103        !(self.may_recv_appdata && self.may_send_appdata)
104    }
105
106    /// Checks if the connection wants to write data.
107    ///
108    /// # Returns
109    /// `true` if there is data to write, `false` otherwise.
110    pub(crate) fn wants_write(&self) -> bool {
111        !self.sendable_tls.is_empty()
112    }
113
114    /// Checks if the connection wants to read data.
115    ///
116    /// # Returns
117    /// `true` if there is data to read, `false` otherwise.
118    pub(crate) fn wants_read(&self) -> bool {
119        self.received_plaintext.is_empty()
120            && !self.has_received_close
121            && (self.may_send_appdata || self.sendable_tls.is_empty())
122    }
123
124    /// Writes TLS data to the given writer.
125    ///
126    /// # Arguments
127    /// * `wr` - The writer to write to.
128    ///
129    /// # Returns
130    /// The number of bytes written or an error.
131    pub(crate) fn write_tls(&mut self, wr: &mut dyn io::Write) -> io::Result<usize> {
132        self.sendable_tls.write_to(wr)
133    }
134
135    /// Processes the main protocol logic.
136    ///
137    /// # Arguments
138    /// * `message` - The message to process.
139    /// * `state` - The current state.
140    /// * `sendable_plaintext` - The buffer for sendable plaintext.
141    ///
142    /// # Returns
143    /// The next state or an error.
144    fn process_main_protocol(
145        &mut self,
146        message: Message<'_>,
147        mut state: Box<dyn State>,
148        sendable_plaintext: &mut [u8],
149    ) -> Result<Box<dyn State>, Error> {
150        if self.may_send_appdata
151            && matches!(
152                message.payload,
153                MessagePayload::HandshakePayload(HandshakePayload::ClientHello(_))
154            )
155        {
156            self.send_warning(AlertDescription::NoRenegotiation);
157            return Ok(state);
158        };
159
160        let mut cx = Context { state: self };
161        match state.handle(&mut cx, message) {
162            Ok(state) => Ok(state),
163            Err(e) => todo!("{e:?}"),
164        }
165    }

167    /// Starts encryption with the given connection secrets.
168    ///
169    /// # Arguments
170    /// * `secrets` - The connection secrets.
171    pub(crate) fn start_encryption(&mut self, secrets: &ConnectionSecrets) {
172        let (dec, enc) = secrets.make_cipher_pair();
173        self.record_layer.prepare_encrypter(enc);
174        self.record_layer.prepare_decrypter(dec);
175    }

177    /// Queues a TLS message for sending.
178    ///
179    /// # Arguments
180    /// * `msg` - The message to queue.
181    fn queue_tls_message(&mut self, msg: OutboundOpaqueMessage) {
182        self.sendable_tls.append(&mut msg.encode())
183    }

185    /// Sends a TLS message.
186    ///
187    /// # Arguments
188    /// * `msg` - The message to send.
189    /// * `must_encrypt` - Indicates if the message must be encrypted.
190    pub fn send_message(&mut self, msg: Message<'_>, must_encrypt: bool) {
191        if !must_encrypt {
192            self.fragmenter
193                .fragment_message(&msg.into())
194                .for_each(|fragment| self.queue_tls_message(fragment.to_unencrypted_opaque()));
195            return;
196        }
197
198        self.fragmenter
199            .fragment_message(&msg.into())
200            .for_each(|fragment| self.send_single_fragment(fragment));
201    }
202
203    /// Sends a single fragment of a TLS message.
204    ///
205    /// # Arguments
206    /// * `m` - The fragment to send.
207    fn send_single_fragment(&mut self, m: OutboundPlainMessage<'_>) {
208        let em = self.record_layer.encrypt(m);
209        self.queue_tls_message(em);
210    }

227    /// * `payload` - The plaintext data to send.
228    ///
229    /// # Returns
230    /// The number of bytes sent.
231    fn send_plain_non_buffering(&mut self, payload: &[u8]) -> usize {
232        if payload.is_empty() {
233            return 0;
234        }

239    /// Encrypts and sends application data.
240    ///
241    /// # Arguments
242    /// * `payload` - The application data to send.
243    ///
244    /// # Returns
245    /// The number of bytes sent.
246    fn send_appdata_encrypt(&mut self, payload: &[u8]) -> usize {
247        self.fragmenter
248            .fragment_payload(
249                ContentType::ApplicationData,
250                ProtocolVersion::TLSv1_2,
251                payload,
252            )
253            .for_each(|fragment| self.send_single_fragment(fragment));
254        payload.len()
255    }

263    /// Sends a warning alert.
264    ///
265    /// # Arguments

267    pub fn send_warning(&mut self, description: AlertDescription) {
268        warn!("Sending warning: {description:?}");
269        let msg = Message::build_alert(AlertLevel::Warning, description);
270        self.send_message(msg, self.record_layer.should_encrypt());

273    /// Sends a fatal alert and returns an error.
274    ///
275    /// # Arguments
276    /// * `description` - The alert description.
277    /// * `err` - The error to send.
278    ///
279    /// # Returns
280    /// The error.
281    pub fn send_fatal(&mut self, description: AlertDescription, err: impl Into<Error>) -> Error {
282        error!("Sending error: {description:?}");
283        let m = Message::build_alert(AlertLevel::Fatal, description);
284        self.send_message(m, self.record_layer.should_encrypt());
285        err.into()
286    }
287
288    /// Processes an alert message.
289    ///
290    /// # Arguments
291    /// * `alert` - The alert payload.
292    ///
293    /// # Returns
294    /// `Ok` if the alert was processed successfully, otherwise an error.
295    fn process_alert(&mut self, alert: &AlertPayload) -> Result<(), Error> {
296        if let AlertLevel::Unknown(_) = alert.level {
297            return Err(self.send_fatal(
298                AlertDescription::IllegalParameter,
299                Error::AlertReceived(alert.description),
300            ));
301        }
302
303        if self.may_recv_appdata && alert.description == AlertDescription::CloseNotify {
304            self.has_received_close = true;
305            return Ok(());
306        };
307
308        if alert.level == AlertLevel::Warning {
309            warn!("Received Alert {alert:?}");
310            return Ok(());
311        }
312
313        error!("Received fatal alert {alert:?}");
314        Err(Error::AlertReceived(alert.description))
315    }

317    /// Checks the state of the connection when no bytes are available.
318    ///
319    /// # Returns
320    /// `Ok` if the state is valid, otherwise an error.
321    pub(crate) fn check_no_bytes_state(&self) -> io::Result<()> {
322        match (self.has_received_close, self.has_seen_eof) {
323            (true, _) => Ok(()),
324            (false, true) => Err(io::Error::new(
325                io::ErrorKind::UnexpectedEof,
326                "unexpected EOF",
327            )),
328            (false, false) => Err(io::ErrorKind::WouldBlock.into()),
329        }
330    }
331
332    pub fn send_close_notify(&mut self) {
333        self.send_warning(AlertDescription::CloseNotify);
334    }

337/// Represents the core of a TLS connection.
338pub struct ConnectionCore {
339    /// The current state of the connection.
340    state: Result<Box<dyn State>, Error>,
341    /// The TLS state.
342    pub(crate) tls_state: TlsState,
343}
344
345impl ConnectionCore {
346    /// Creates a new `ConnectionCore`.
347    ///
348    /// # Arguments
349    /// * `state` - The initial state of the connection.
350    /// * `tls_state` - The TLS state.
351    ///
352    /// # Returns
353    /// A new `ConnectionCore` instance.
354    pub fn new(state: Box<dyn State>, tls_state: TlsState) -> Self {
355        Self {
356            state: Ok(state),
357            tls_state,
358        }
359    }
360
361    /// Processes new packets from the deframer.
362    ///
363    /// # Arguments
364    /// * `deframer` - The deframer buffer.
365    /// * `sendable_plaintext` - The buffer for sendable plaintext.
366    ///
367    /// # Returns
368    /// `Ok(())` if successful, otherwise an error.
369    fn process_new_packets(
370        &mut self,
371        deframer: &mut VecDeframerBuffer,
372        sendable_plaintext: &mut [u8],
373    ) -> Result<(), Error> {
374        let mut state = match mem::replace(&mut self.state, Err(Error::HandshakeNotCompleate)) {
375            Ok(state) => state,
376            Err(err) => {
377                self.state = Err(err.clone());
378                return Err(err);
379            }
380        };
381
382        loop {
383            let res = self.deframe(deframer.filled_mut());
384
385            let msg_opt = match res {
386                Ok(opt) => opt,
387                Err(_) => todo!(),
388            };
389
390            let Some((msg, size)) = msg_opt else {
391                break;
392            };
393
394            match self.process_msg(msg, state, sendable_plaintext) {
395                Ok(s) => state = s,
396                Err(e) => {
397                    self.state = Err(e.clone());
398                    deframer.discard(size);
399                    return Err(e);
400                }
401            }
402
403            if self.has_received_close {
404                deframer.discard(deframer.filled().len());
405                break;
406            }
407
408            deframer.discard(size);
409        }
410
411        self.state = Ok(state);
412        Ok(())
413    }

415    /// Deframes a buffer into a message.
416    ///
417    /// # Arguments
418    /// * `buffer` - The buffer to deframe.
419    ///
420    /// # Returns
421    /// An optional tuple containing the inbound plain message and its size, or an error.
422    fn deframe<'b>(
423        &mut self,
424        buffer: &'b mut [u8],
425    ) -> Result<Option<(InboundPlainMessage<'b>, usize)>, Error> {
426        let is_handshaking = self.is_handshaking();
427
428        let mut iter = DeframerIter::new(buffer);
429        let message = match iter.next().transpose() {
430            Ok(Some(message)) => message,
431            Ok(None) => return Ok(None),
432            Err(e) => return Err(self.handle_deframe_error(e)),
433        };
434
435        let allowed_plaintext = match message.typ {
436            // CCS is always plaintext
437            ContentType::ChangeCipherSpec => true,
438            // Handshake Finished Message should not be plaintext!!
439            ContentType::Handshake => !self.record_layer.should_decrypt(),
440            ContentType::Alert => is_handshaking,
441            _ => false,
442        };
443
444        if allowed_plaintext {
445            return Ok(Some((message.into(), iter.consumed())));
446        }
447
448        let message = match self.record_layer.decrypt(message) {
449            Ok(message) => message,
450            Err(e) => todo!("handle decryption error {:?}", e),
451        };
452
453        Ok(Some((message, iter.consumed())))
454    }
455
456    /// Processes a message.
457    ///
458    /// # Arguments
459    /// * `msg` - The inbound plain message.
460    /// * `state` - The current state.
461    /// * `sendable_plaintext` - The buffer for sendable plaintext.
462    ///
463    /// # Returns
464    /// The next state or an error.
465    fn process_msg(
466        &mut self,
467        msg: InboundPlainMessage,
468        state: Box<dyn State>,
469        sendable_plaintext: &mut [u8],
470    ) -> Result<Box<dyn State>, Error> {
471        let msg = match Message::try_from(msg) {
472            Ok(msg) => msg,
473            Err(e) => todo!("{:?}", e),
474        };
475
476        if let MessagePayload::Alert(alert) = &msg.payload {
477            self.process_alert(alert)?;
478            return Ok(state);
479        }
480
481        self.process_main_protocol(msg, state, sendable_plaintext)
482    }
483
484    /// Handles deframe errors.
485    ///
486    /// # Arguments
487    /// * `err` - The error to handle.
488    ///
489    /// # Returns
490    /// The handled error.
491    fn handle_deframe_error(&mut self, err: Error) -> Error {
492        match err {
493            Error::InvalidMessage(err) => self.send_fatal(AlertDescription::DecodeError, err),
494            Error::PeerSendOversizedRecord => {
495                self.send_fatal(AlertDescription::RecordOverflow, err)
496            }
497            Error::DecryptError => self.send_fatal(AlertDescription::DecryptError, err),
498            error => error,
499        }
500    }
501}

592    /// Reads TLS data from the given reader.
593    ///
594    /// # Arguments
595    /// * `r` - The reader to read from.
596    ///
597    /// # Returns
598    /// The number of bytes read or an error.
599    pub(crate) fn read_tls(&mut self, r: &mut dyn io::Read) -> io::Result<usize> {
600        let res = self.deframer_buffer.read(r, self.is_handshaking());
601        if let Ok(0) = res {
602            self.has_seen_eof = true;
603        }
604        res
605    }
606
607    /// Completes I/O operations for the connection.
608    ///
609    /// # Arguments
610    /// * `io` - The I/O object to read from and write to.
611    ///
612    /// # Returns
613    /// A tuple containing the number of bytes read and written, or an error.
614    pub(crate) fn complete_io<T: io::Read + io::Write>(
615        &mut self,
616        io: &mut T,
617    ) -> Result<(usize, usize), io::Error> {
618        let mut eof = false;
619        let mut read_len = 0;
620        let mut write_len = 0;
621
622        loop {
623            if !self.wants_read() && !self.wants_write() {
624                return Ok((read_len, write_len));
625            }
626
627            while self.wants_write() {
628                match self.write_tls(io)? {
629                    0 => {
630                        io.flush()?;
631                        return Ok((read_len, write_len));
632                    }
633                    n => write_len += n,
634                }
635            }
636            io.flush()?;
637
638            if !self.is_handshaking() && write_len > 0 {
639                return Ok((read_len, write_len));
640            }
641
642            while !eof && self.wants_read() {
643                let bytes_read = match self.read_tls(io) {
644                    Ok(0) => {
645                        eof = true;
646                        Some(0)
647                    }
648                    Ok(n) => {
649                        read_len += n;
650                        Some(n)
651                    }
652                    Err(ref err) if err.kind() == io::ErrorKind::Interrupted => None,
653                    Err(err) => return Err(err),
654                };
655
656                if bytes_read.is_some() {
657                    break;
658                }
659            }
660
661            match self.process_new_packets() {
662                Ok(_) => {}
663                Err(e) => {
664                    _ = self.write_tls(io);
665                    _ = io.flush();
666
667                    return Err(io::Error::new(io::ErrorKind::InvalidData, e));
668                }
669            }
670        }
671    }

673    /// Processes new packets.

675    /// # Returns

679        self.core

683    /// Retrieves the first handshake message.
684    ///
685    /// # Returns
686    /// An optional handshake message or an error.
687    pub(crate) fn first_handshake_message(&mut self) -> Result<Option<Message<'static>>, Error> {
688        let (res) = self
689            .core
690            .deframe(self.deframer_buffer.filled_mut())
691            .map(|opt| opt.map(|(pm, len)| Message::try_from(pm).map(|m| (m.into_owned(), len))));
692
693        match res? {
694            Some(Ok((msg, len))) => {
695                self.deframer_buffer.discard(len);
696                Ok(Some(msg))
697            }
698            Some(Err(err)) => Err(self.send_fatal(AlertDescription::DecodeError, err)),
699            None => Ok(None),
700        }
701    }

712/// Trait for writing data to an I/O object.
713pub trait WriteTo {
714    /// Writes data to the given writer.
715    ///
716    /// # Arguments
717    /// * `wr` - The writer to write to.
718    ///
719    /// # Returns
720    /// The number of bytes written or an error.
721    fn write_to(&mut self, wr: &mut dyn io::Write) -> io::Result<usize>;
722}
723
724impl WriteTo for Vec<u8> {
725    /// Writes data from the vector to the given writer.
726    ///
727    /// # Arguments
728    /// * `wr` - The writer to write to.
729    ///
730    /// # Returns
731    /// The number of bytes written or an error.
732    fn write_to(&mut self, wr: &mut dyn io::Write) -> io::Result<usize> {
733        match wr.write(self) {
734            Ok(len) => {
735                self.drain(..len);
736                Ok(len)
737            }
738            Err(e) => Err(e),
739        }
740    }
