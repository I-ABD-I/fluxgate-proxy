11#[derive(Debug)]
12struct Servers(Arc<RwLock<HashMap<Upstream, f32>>>);
13
14impl Servers {
15    fn new(upstreams: Vec<Upstream>) -> Self {
16        Self(Arc::new(RwLock::new(
17            upstreams.into_iter().map(|u| (u, 0.0)).collect(),
18        )))
19    }
20
21    fn dup(&self) -> Self {
22        Self(self.0.clone())
23    }
24}
25
26impl Deref for Servers {
27    type Target = Arc<RwLock<HashMap<Upstream, f32>>>;
28
29    fn deref(&self) -> &Self::Target {
30        &self.0
31    }
32}
33
34#[derive(Debug)]
35pub struct ResourceBased {
36    servers: Servers,
37}
38
39impl From<Vec<Upstream>> for ResourceBased {
40    fn from(value: Vec<Upstream>) -> Self {
41        let servers = Servers::new(value);
42        async_std::task::spawn(updater(servers.dup()));
43
44        Self { servers }
45    }
46}
47
48impl LoadBalancer for ResourceBased {
49    fn get_upstream(&self) -> Option<Upstream> {
50        self.servers
51            .read_blocking()
52            .iter()
53            .min_by(|a, b| a.1.partial_cmp(b.1).unwrap_or(Ordering::Greater))
54            .map(|(k, _)| *k)
55    }
56}

58async fn updater(servers: Servers) -> ! {
59    let udp = UdpSocket::bind("0.0.0.0:0").await.unwrap();
60
61    let mut send_buf = Vec::with_capacity(1);
62    Message::Connect.encode(&mut send_buf);
63
64    for server in servers.read().await.keys() {
65        let mut addr = server.addr;
66        addr.set_port(2749);
67        udp.send_to(send_buf.as_slice(), addr).await.unwrap();
68        async_std::task::sleep(std::time::Duration::from_secs(1)).await; // create a delay between messages such that both servers wont have same usage
69    }
70
71    send_buf.clear();
72    Message::Ack.encode(&mut send_buf);
73    loop {
74        let mut buf = [0u8; 5];
75        let (size, addr) = udp.recv_from(&mut buf).await.unwrap();
76        let msg = Message::read(&mut Reader::from(&buf[..size]));
77        let Ok(msg) = msg else { continue };
78        use Message::*;
79        match msg {
80            Ack => {}
81            Metrics(m) => {
82                let _ = udp.send_to(&send_buf, addr).await.unwrap();
83                let mut servers = servers.write().await;
84                if let Some((_, cpu)) = servers.iter_mut().find(|s| s.0.addr.ip() == addr.ip()) {
85                    *cpu = m.cpu;
86                }
87            }
88            _ => {}
89        }
90    }
91}
