9/// A builder for composing layers and services.
10pub struct ServiceBuilder<L> {
11    inner: L,
12}
13
14impl Default for ServiceBuilder<IdentityLayer> {
15    /// Creates a new `ServiceBuilder` with the `Identity` layer.
16    fn default() -> Self {
17        Self::new()
18    }
19}
20
21impl ServiceBuilder<IdentityLayer> {
22    /// Creates a new `ServiceBuilder` with the `Identity` layer.
23    ///
24    /// # Returns
25    /// A new instance of `ServiceBuilder` with the `Identity` layer.
26    pub const fn new() -> Self {
27        Self {
28            inner: IdentityLayer,
29        }
30    }
31}
32
33impl<L> ServiceBuilder<L> {
34    /// Adds a layer to the `ServiceBuilder`.
35    ///
36    /// # Arguments
37    /// * `layer` - The layer to be added.
38    ///
39    /// # Returns
40    /// A new `ServiceBuilder` with the added layer.
41    pub fn layer<T>(self, layer: T) -> ServiceBuilder<util::Stack<T, L>> {
42        ServiceBuilder {
43            inner: Stack::new(layer, self.inner),
44        }
45    }
46
47    /// Adds an optional layer to the `ServiceBuilder`.
48    ///
49    /// # Arguments
50    /// * `layer` - An optional layer to be added.
51    ///
52    /// # Returns
53    /// A new `ServiceBuilder` with the added layer if present, otherwise with the `Identity` layer.
54    pub fn option_layer<T>(
55        self,
56        layer: Option<T>,
57    ) -> ServiceBuilder<util::Stack<Either<T, IdentityLayer>, L>> {
58        let inner = match layer {
59            Some(layer) => Either::Left(layer),
60            None => Either::Right(IdentityLayer),
61        };
62        ServiceBuilder {
63            inner: Stack::new(inner, self.inner),
64        }
65    }
66
67    /// Wraps a service with the composed layers.
68    ///
69    /// # Arguments
70    /// * `service` - The service to be wrapped.
71    ///
72    /// # Returns
73    /// The wrapped service.
74    pub fn service<S>(&self, service: S) -> L::Service
75    where
76        L: Layer<S>,
77    {
78        self.inner.layer(service)
79    }
80
81    /// Wraps a function as a service with the composed layers.
82    ///
83    /// # Arguments
84    /// * `f` - The function to be wrapped as a service.
85    ///
86    /// # Returns
87    /// The wrapped service.
88    pub fn service_fn<F>(&self, f: F) -> L::Service
89    where
90        L: Layer<ServiceFn<F>>,
91    {
92        self.inner.layer(ServiceFn::new(f))
93    }
94
95    pub fn build<T>(&self) -> L::Service
96    where
97        L: Layer<IdentityService<T>>,
98    {
99        self.inner.layer(IdentityService::default())
100    }
101}
